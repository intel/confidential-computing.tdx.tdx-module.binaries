struct tdx_blob
{
	_u16 version;              // Version number
	_u16 checksum;             // Checksum of the entire blob should be zero
	_u8  signature[8];         // Must be "TDX-BLOB"
	_u32 sigstruct_nr_pages;   // Number of 4KB pages occupied by sigstruct data
	_u32 module_nr_pages;      // Number of 4KB pages occupied by module data
	_u8  reserved[4076];       // Reserved space to pad the header to 4KB
	_u8  sigstruct[];          // intel_tdx_module.so.sigstruct, 4KB aligned, zero-padded
	_u8  module[];             // intel_tdx_module.bin, 4KB aligned
}


** The <version> field is defined to be 2.0 (0x200).
** For sigstruct files that are 2KB, <sigstruct_nr_pages> is equal to one.
** <module_nr_pages> is the number of 4KB pages required to hold the module binary
** (i.e. ceil(module_size / 4096)).
** All reserved fields must be set to zero.
